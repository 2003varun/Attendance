        /* =========================================================
           GLOBAL DATA
        ========================================================= */

        let rawData = [];
        let filteredData = [];
        let rawLines = [];
        let charts = {};


        /* =========================================================
           CONFIGURATION
        ========================================================= */

        const WORK_START = 9 * 60 + 30; // 09:30 AM Work Start

        // Lunch starts at 14:00 (2:00 PM) in your attendance setup.
        const LUNCH_START = 14 * 60;

        // Lunch ends at 14:30 (2:30 PM) - Check-in after 14:30 is considered late afternoon attendance.
        const LUNCH_END = 14 * 60 + 30;

        // Used for deciding morning/afternoon half-day.
        const HALF_DAY_CUTOFF = 13 * 60 + 30;


        /* =========================================================
           BASIC HELPERS
        ========================================================= */

        function escapeHtml(v) {

            return String(v ?? "").replace(
                /[&<>"']/g,
                function (m) {

                    return {
                        "&": "&amp;",
                        "<": "&lt;",
                        ">": "&gt;",
                        '"': "&quot;",
                        "'": "&#039;"
                    }[m];

                }
            );

        }


        function norm(s) {

            return String(s || "")
                .trim()
                .toLowerCase()
                .replace(/[^a-z0-9]/g, "");

        }


        function minutesToTime(m) {

            if (m === null || m === undefined) {
                return "";
            }

            return String(
                Math.floor(m / 60)
            ).padStart(2, "0")
                + ":" +
                String(
                    m % 60
                ).padStart(2, "0");

        }


        function formatDate(date) {

            if (!date) {
                return "";
            }

            const p = date.split("-");

            if (p.length !== 3) {
                return date;
            }

            const months = [
                "JAN", "FEB", "MAR", "APR", "MAY", "JUN",
                "JUL", "AUG", "SEP", "OCT", "NOV", "DEC"
            ];

            const y = p[0];
            const m = Number(p[1]);
            const d = p[2];

            return String(d).padStart(2, "0")
                + "-" +
                months[m - 1]
                + "-" +
                y;

        }


        function getDayName(date) {

            if (!date) {
                return "";
            }

            const d = new Date(date + "T12:00:00");

            if (isNaN(d)) {
                return "";
            }

            return d.toLocaleDateString(
                "en-US",
                { weekday: "long" }
            );

        }


        function getDateFromTimestamp(timestamp) {

            const match = String(timestamp).match(
                /(\d{4})-(\d{2})-(\d{2})/
            );

            if (!match) {
                return "";
            }

            return `${match[1]}-${match[2]}-${match[3]}`;

        }


        function getTimeFromTimestamp(timestamp) {

            const match = String(timestamp).match(
                /(\d{1,2}):(\d{2})(?::(\d{2}))?/
            );

            if (!match) {
                return null;
            }

            return (
                Number(match[1]) * 60 +
                Number(match[2])
            );

        }


        /* =========================================================
           RAW TXT PARSER
        ========================================================= */

        /*
        
        Expected raw line:
        
        378    2026-08-01 09:27:45    1    0    3    0
        
        The parser does NOT depend on tabs specifically.
        
        It also accepts multiple spaces.
        
        */

        function parseRawTxt(text) {

            const lines = text
                .replace(/\r\n/g, "\n")
                .replace(/\r/g, "\n")
                .split("\n");

            rawLines = lines;

            const punches = [];

            lines.forEach((line, index) => {

                const clean = line.trim();

                if (!clean) {
                    return;
                }

                /*
                 Find:
              
                 employee ID
                 YYYY-MM-DD
                 HH:MM:SS
                */

                const match = clean.match(
                    /^\s*(\d+)\s+(\d{4}-\d{2}-\d{2})\s+(\d{1,2}:\d{2}(?::\d{2})?)\s*(.*)$/
                );

                if (!match) {
                    return;
                }

                const employeeId = match[1];
                const date = match[2];
                const time = match[3];
                const rest = match[4] || "";

                const minutes = getTimeFromTimestamp(
                    `${date} ${time}`
                );

                if (minutes === null) {
                    return;
                }

                punches.push({

                    employeeId: employeeId,

                    date: date,

                    time: time,

                    minutes: minutes,

                    raw: clean,

                    extra: rest,

                    sourceLine: index + 1

                });

            });


            return punches;

        }


        /* =========================================================
           GROUP RAW PUNCHES
        ========================================================= */

        function buildAttendanceFromRaw(punches) {

            const groups = new Map();

            punches.forEach(p => {

                const key =
                    `${p.employeeId}|${p.date}`;

                if (!groups.has(key)) {

                    groups.set(
                        key,
                        {
                            employeeId: p.employeeId,
                            employeeName: p.employeeId,
                            date: p.date,
                            punches: []
                        }
                    );

                }

                groups.get(key).punches.push(p);

            });


            const result = [];

            groups.forEach(group => {

                group.punches.sort(
                    (a, b) => a.minutes - b.minutes
                );

                const punchCount = group.punches.length;
                let firstLoginMin = null;
                let afternoonLogoutMin = null;
                let afternoonLoginMin = null;
                let lastLogoutMin = null;

                if (punchCount === 1) {

                    const p0 = group.punches[0].minutes;
                    if (p0 < LUNCH_START) {
                        firstLoginMin = p0;
                    } else if (p0 <= LUNCH_END + 15) {
                        // Checkin within or around 2:30 PM
                        afternoonLoginMin = p0;
                        lastLogoutMin = p0;
                    } else {
                        lastLogoutMin = p0;
                    }

                } else if (punchCount === 2) {

                    const p0 = group.punches[0].minutes;
                    const p1 = group.punches[1].minutes;

                    if (p0 < 12 * 60) {
                        firstLoginMin = p0;
                        if (p1 >= LUNCH_START && p1 <= LUNCH_END + 15) {
                            // Logout at/after 2:00 PM
                            afternoonLogoutMin = p1;
                            lastLogoutMin = p1;
                        } else {
                            lastLogoutMin = p1;
                        }
                    } else if (p0 >= LUNCH_START) {
                        // Checkin in afternoon (within or after 2:30 PM)
                        afternoonLoginMin = p0;
                        lastLogoutMin = p1;
                    } else {
                        firstLoginMin = p0;
                        lastLogoutMin = p1;
                    }

                } else if (punchCount === 3) {

                    const p0 = group.punches[0].minutes;
                    const p1 = group.punches[1].minutes;
                    const p2 = group.punches[2].minutes;

                    firstLoginMin = p0;
                    lastLogoutMin = p2;

                    // If middle punch is logout at or after 2:00 PM (14:00 - 14:15)
                    if (p1 >= LUNCH_START && p1 <= 14 * 60 + 15) {
                        afternoonLogoutMin = p1;
                    } else if (p1 > 14 * 60 + 15 && p1 <= 16 * 60) {
                        // Middle punch is checkin within or after 2:30 PM
                        afternoonLoginMin = p1;
                    } else if (p1 < LUNCH_START) {
                        afternoonLogoutMin = p1;
                    } else {
                        afternoonLoginMin = p1;
                    }

                } else if (punchCount >= 4) {

                    firstLoginMin = group.punches[0].minutes;
                    lastLogoutMin = group.punches[punchCount - 1].minutes;

                    const midPunches = group.punches.slice(1, -1);

                    // 1. Afternoon Check-out: Logout at or after 2:00 PM (14:00)
                    const logoutAfter2Candidates = midPunches.filter(p => p.minutes >= LUNCH_START - 20 && p.minutes <= LUNCH_START + 25);
                    if (logoutAfter2Candidates.length > 0) {
                        afternoonLogoutMin = logoutAfter2Candidates[0].minutes;
                    } else {
                        const earlyMid = midPunches.filter(p => p.minutes < LUNCH_START + 25);
                        if (earlyMid.length > 0) {
                            afternoonLogoutMin = earlyMid[earlyMid.length - 1].minutes;
                        }
                    }

                    // 2. Afternoon Check-in: Return / Check-in within 2:30 PM (or after 2:30 PM)
                    const checkinCandidates = midPunches.filter(p =>
                        p.minutes >= LUNCH_START &&
                        (afternoonLogoutMin === null || p.minutes > afternoonLogoutMin)
                    );
                    if (checkinCandidates.length > 0) {
                        afternoonLoginMin = checkinCandidates[0].minutes;
                    }

                    if (afternoonLogoutMin === null && afternoonLoginMin === null && midPunches.length >= 2) {
                        afternoonLogoutMin = midPunches[0].minutes;
                        afternoonLoginMin = midPunches[1].minutes;
                    }

                }


                let status = "FULL DAY PRESENT";

                if (punchCount === 0 || (firstLoginMin === null && afternoonLoginMin === null && lastLogoutMin === null)) {

                    status = "ABSENT";

                } else if (
                    new Date(
                        group.date + "T12:00:00"
                    ).getDay() === 0
                ) {

                    status = "SUNDAY / WEEKLY OFF";

                } else if (
                    firstLoginMin !== null &&
                    firstLoginMin < 12 * 60 &&
                    lastLogoutMin !== null &&
                    lastLogoutMin < HALF_DAY_CUTOFF &&
                    afternoonLoginMin === null
                ) {

                    status = "HALF DAY - MORNING";

                } else if (
                    (firstLoginMin === null || firstLoginMin >= HALF_DAY_CUTOFF) &&
                    afternoonLoginMin !== null
                ) {

                    status = "HALF DAY - AFTERNOON";

                }


                /*
                 Calculate morning late minutes (Work starts at 09:30).
                */
                let morningLateMinutes = 0;

                if (
                    firstLoginMin !== null &&
                    firstLoginMin > WORK_START &&
                    status !== "SUNDAY / WEEKLY OFF" &&
                    status !== "ABSENT"
                ) {

                    morningLateMinutes =
                        firstLoginMin - WORK_START;

                }


                /*
                 Calculate afternoon late minutes (Lunch ends at 14:30 / 2:30 PM).
                 Check-in after 2:30 PM is counted as late attendance.
                */
                let afternoonLateMinutes = 0;

                if (
                    afternoonLoginMin !== null &&
                    afternoonLoginMin > LUNCH_END &&
                    status !== "SUNDAY / WEEKLY OFF" &&
                    status !== "ABSENT"
                ) {

                    afternoonLateMinutes =
                        afternoonLoginMin - LUNCH_END;

                }


                const totalLateMinutes =
                    morningLateMinutes + afternoonLateMinutes;


                let attendanceValue = 1;

                if (
                    status === "HALF DAY - MORNING" ||
                    status === "HALF DAY - AFTERNOON"
                ) {

                    attendanceValue = 0.5;

                }

                if (status === "ABSENT" || status === "SUNDAY / WEEKLY OFF") {

                    attendanceValue = 0;

                }


                result.push({

                    _index: result.length + 1,

                    date: group.date,

                    day: getDayName(group.date),

                    employeeId: group.employeeId,

                    employeeName: group.employeeName,

                    status: status,

                    attendanceValue: attendanceValue,

                    lateMinutes: totalLateMinutes,

                    morningLateMinutes: morningLateMinutes,

                    afternoonLateMinutes: afternoonLateMinutes,

                    lateFine: 0,

                    firstLogin: firstLoginMin !== null
                        ? minutesToTime(firstLoginMin)
                        : "",

                    afternoonLogout: afternoonLogoutMin !== null
                        ? minutesToTime(afternoonLogoutMin)
                        : "",

                    afternoonLogin: afternoonLoginMin !== null
                        ? minutesToTime(afternoonLoginMin)
                        : "",

                    lastLogout: lastLogoutMin !== null
                        ? minutesToTime(lastLogoutMin)
                        : "",

                    punchCount: punchCount,

                    punchTimes: group.punches
                        .map(x => x.time)
                        .join(", "),

                    rawPunches: group.punches
                        .map(x => x.raw)
                        .join("\n")

                });

            });


            result.sort((a, b) => {

                if (a.date !== b.date) {
                    return a.date.localeCompare(b.date);
                }

                return String(a.employeeId)
                    .localeCompare(
                        String(b.employeeId),
                        undefined,
                        { numeric: true }
                    );

            });


            return result;

        }

        function csvParse(text) {

            const rows = [];
            let row = [];
            let cell = "";
            let quoted = false;

            for (
                let i = 0;
                i < text.length;
                i++
            ) {

                const c = text[i];
                const n = text[i + 1];

                if (c === '"') {

                    if (quoted && n === '"') {

                        cell += '"';
                        i++;

                    } else {

                        quoted = !quoted;

                    }

                }

                else if (c === "," && !quoted) {

                    row.push(cell);
                    cell = "";

                }

                else if (
                    (c === "\n" || c === "\r") &&
                    !quoted
                ) {

                    if (c === "\r" && n === "\n") {
                        i++;
                    }

                    row.push(cell);

                    if (
                        row.some(
                            x => String(x).trim() !== ""
                        )
                    ) {

                        rows.push(row);

                    }

                    row = [];
                    cell = "";

                }

                else {

                    cell += c;

                }

            }

            if (
                cell.length ||
                row.length
            ) {

                row.push(cell);
                rows.push(row);

            }

            return rows;

        }


        function canonicalStatus(v) {

            const s =
                String(v || "")
                    .trim()
                    .toUpperCase();

            if (s.includes("MORNING"))
                return "HALF DAY - MORNING";

            if (
                s.includes("AFTERNOON") ||
                s.includes("NOON")
            )
                return "HALF DAY - AFTERNOON";

            if (
                s.includes("FULL") ||
                s === "P" ||
                s === "PRESENT"
            )
                return "FULL DAY PRESENT";

            if (
                s.includes("ABSENT") ||
                s === "A"
            )
                return "ABSENT";

            if (
                s.includes("SUNDAY") ||
                s.includes("WEEKLY OFF") ||
                s === "OFF"
            )
                return "SUNDAY / WEEKLY OFF";

            return s || "UNKNOWN";

        }


        function parseDateValue(v) {

            v = String(v || "").trim();

            if (!v) {
                return "";
            }

            if (
                /^\d{4}-\d{2}-\d{2}$/.test(v)
            ) {
                return v;
            }

            let m =
                v.match(
                    /^(\d{1,2})[-\/ ]([A-Za-z]{3,9}|\d{1,2})[-\/ ](\d{2,4})$/
                );

            if (m) {

                let d = m[1];
                let mon = m[2];
                let y = m[3];

                if (y.length === 2) {
                    y = "20" + y;
                }

                const months = [
                    "jan", "feb", "mar", "apr", "may", "jun",
                    "jul", "aug", "sep", "oct", "nov", "dec"
                ];

                let mn =
                    months.findIndex(
                        x => mon.toLowerCase().startsWith(x)
                    ) + 1;

                if (!mn) {
                    mn = Number(mon);
                }

                if (mn) {

                    return `${y}-${String(mn).padStart(2, "0")}-${String(d).padStart(2, "0")}`;

                }

            }

            const dt = new Date(v);

            if (isNaN(dt)) {
                return "";
            }

            return dt.toISOString().slice(0, 10);

        }


        function pick(obj, ...names) {

            for (const n of names) {

                if (
                    obj[n] !== undefined &&
                    obj[n] !== ""
                ) {

                    return obj[n];

                }

            }

            return "";

        }


        function loadCSV(file) {

            const reader = new FileReader();

            reader.onload = function (e) {

                const rows = csvParse(e.target.result);

                if (rows.length < 2) {

                    alert(
                        "The CSV file has no data."
                    );

                    return;

                }


                const headers =
                    rows[0].map(
                        (h, i) => norm(h) || `col${i}`
                    );


                rawData =
                    rows.slice(1)
                        .map((arr, index) => {

                            const o = {};

                            headers.forEach(
                                (h, i) => {
                                    o[h] = (arr[i] || "").trim();
                                }
                            );


                            const date =
                                parseDateValue(
                                    pick(o, "date")
                                );


                            const status =
                                canonicalStatus(
                                    pick(
                                        o,
                                        "attendancestatus",
                                        "status",
                                        "attendance"
                                    )
                                );


                            const firstLogin =
                                pick(o, "firstlogin", "login1", "login");

                            const afternoonLogout =
                                pick(o, "afternooncheckout", "afternoonlogout", "lunchout", "logout1", "afternoonout");

                            const afternoonLogin =
                                pick(o, "afternooncheckin", "afternoonlogin", "lunchin", "login2", "afternoonin");

                            const lastLogout =
                                pick(o, "lastlogout", "logout2", "logout");

                            const firstLoginMin = timeToMinutes(firstLogin);
                            const afternoonLoginMin = timeToMinutes(afternoonLogin);

                            let morningLate = 0;
                            if (
                                firstLoginMin !== null &&
                                firstLoginMin > WORK_START &&
                                status !== "SUNDAY / WEEKLY OFF" &&
                                status !== "ABSENT"
                            ) {
                                morningLate = firstLoginMin - WORK_START;
                            }

                            let afternoonLate = 0;
                            if (
                                afternoonLoginMin !== null &&
                                afternoonLoginMin > LUNCH_END &&
                                status !== "SUNDAY / WEEKLY OFF" &&
                                status !== "ABSENT"
                            ) {
                                afternoonLate = afternoonLoginMin - LUNCH_END;
                            }

                            const explicitLate = Number(
                                pick(
                                    o,
                                    "totallateminutes",
                                    "lateminutes",
                                    "late"
                                )
                            );

                            const lateMinutes = !isNaN(explicitLate) && explicitLate > 0
                                ? explicitLate
                                : (morningLate + afternoonLate);

                            return {

                                _index: index + 2,

                                date: date,

                                day:
                                    pick(o, "day") ||
                                    getDayName(date),

                                employeeId:
                                    pick(
                                        o,
                                        "employeeid",
                                        "empid",
                                        "employee"
                                    ),

                                employeeName:
                                    pick(
                                        o,
                                        "employeename",
                                        "name"
                                    ),

                                status: status,

                                attendanceValue:
                                    Number(
                                        pick(
                                            o,
                                            "attendancevalue",
                                            "value"
                                        )
                                    ) || (status === "FULL DAY PRESENT" ? 1 : status.includes("HALF DAY") ? 0.5 : 0),

                                lateMinutes: lateMinutes,

                                morningLateMinutes: morningLate,

                                afternoonLateMinutes: afternoonLate,

                                lateFine:
                                    Number(
                                        pick(
                                            o,
                                            "latefineinr",
                                            "latefine",
                                            "fine"
                                        )
                                    ) || 0,

                                firstLogin: firstLogin,

                                afternoonLogout: afternoonLogout,

                                afternoonLogin: afternoonLogin,

                                lastLogout: lastLogout,

                                punchCount:
                                    Number(
                                        pick(o, "punchcount")
                                    ) || 0,

                                punchTimes:
                                    pick(o, "punchtimes")

                            };

                        })
                        .filter(
                            x => x.employeeId || x.employeeName
                        );


                filteredData = [...rawData];


                localStorage.setItem(
                    "attendanceRawTXTDataV3",
                    JSON.stringify(rawData)
                );


                fileInfo.textContent =
                    `Loaded CSV: ${file.name} • ${rawData.length} records`;


                sideStatus.textContent =
                    `${file.name.slice(0, 22).toUpperCase()} LOADED`;


                populateEmployees();

                resetFilters();

                renderAll();

                showPage("dashboard");

            };

            reader.readAsText(file);

        }

        function excelDate(v) {

            if (
                v === null ||
                v === undefined ||
                String(v).trim() === ""
            ) {
                return "";
            }


            if (
                v instanceof Date &&
                !isNaN(v.getTime())
            ) {

                return `${v.getFullYear()}-${String(v.getMonth() + 1).padStart(2, "0")
                    }-${String(v.getDate()).padStart(2, "0")
                    }`;

            }


            if (typeof v === "number") {

                if (!Number.isFinite(v) || v < 1) {
                    return "";
                }

                const p =
                    XLSX.SSF.parse_date_code(
                        Math.floor(v)
                    );

                if (
                    p &&
                    p.y >= 2000 &&
                    p.y <= 2100
                ) {

                    return `${p.y}-${String(p.m).padStart(2, "0")
                        }-${String(p.d).padStart(2, "0")
                        }`;

                }

                return "";

            }


            return parseDateValue(v);

        }


        function timeToMinutes(v) {

            if (
                v === null ||
                v === undefined ||
                v === ""
            ) {
                return null;
            }


            if (v instanceof Date) {

                return (
                    v.getHours() * 60 +
                    v.getMinutes()
                );

            }


            if (
                typeof v === "number" &&
                v >= 0 &&
                v < 1
            ) {

                return Math.round(
                    v * 24 * 60
                );

            }


            const s = String(v).trim();

            const m =
                s.match(
                    /^(\d{1,2}):(\d{2})(?::\d{2})?\s*(AM|PM)?$/i
                );


            if (m) {

                let h = Number(m[1]);
                const mi = Number(m[2]);
                const ap = (m[3] || "").toUpperCase();

                if (ap === "PM" && h < 12) {
                    h += 12;
                }

                if (ap === "AM" && h === 12) {
                    h = 0;
                }

                return h * 60 + mi;

            }


            return null;

        }


        function sheetRowsToObjects(ws) {

            const rows =
                XLSX.utils.sheet_to_json(
                    ws,
                    {
                        header: 1,
                        defval: "",
                        raw: true
                    }
                );


            if (!rows.length) {
                return [];
            }


            const headers =
                rows[0].map(
                    (h, i) => norm(h) || `col${i}`
                );


            return rows
                .slice(1)
                .map(arr => {

                    const o = {};

                    headers.forEach(
                        (h, i) => {
                            o[h] = arr[i] ?? "";
                        }
                    );

                    return o;

                });

        }


        function loadExcel(file) {

            const reader = new FileReader();

            reader.onload = function (e) {

                try {

                    const wb =
                        XLSX.read(
                            new Uint8Array(e.target.result),
                            {
                                type: "array",
                                cellDates: true
                            }
                        );


                    const punchName =
                        wb.SheetNames.find(
                            n => norm(n) === "punchingtime"
                        ) ||
                        wb.SheetNames.find(
                            n => norm(n).includes("punch")
                        );


                    if (!punchName) {

                        alert(
                            "Could not find the Punching time sheet."
                        );

                        return;

                    }


                    const punch =
                        sheetRowsToObjects(
                            wb.Sheets[punchName]
                        );


                    rawData =
                        punch
                            .map((o, i) => {

                                const id =
                                    String(
                                        pick(
                                            o,
                                            "empid",
                                            "employeeid",
                                            "id"
                                        )
                                    ).trim();


                                const name =
                                    String(
                                        pick(
                                            o,
                                            "name",
                                            "nameofemployee",
                                            "employeename"
                                        )
                                    ).trim();


                                const date =
                                    excelDate(
                                        pick(o, "date")
                                    );


                                const logins = [];
                                const logouts = [];


                                for (
                                    let n = 1;
                                    n <= 5;
                                    n++
                                ) {

                                    const li =
                                        n === 1
                                            ? o["login"]
                                            : o[`login${n}`];


                                    const lo =
                                        n === 1
                                            ? o["logout"]
                                            : o[`logout${n}`];


                                    const a =
                                        timeToMinutes(li);


                                    const b =
                                        timeToMinutes(lo);


                                    if (a !== null) {
                                        logins.push(a);
                                    }

                                    if (b !== null) {
                                        logouts.push(b);
                                    }

                                }


                                const first =
                                    logins.length
                                        ? Math.min(...logins)
                                        : null;


                                const last =
                                    logouts.length
                                        ? Math.max(...logouts)
                                        : null;


                                let status =
                                    "FULL DAY PRESENT";


                                if (
                                    first === null &&
                                    last === null
                                ) {

                                    status = "ABSENT";

                                }

                                else if (
                                    date &&
                                    new Date(
                                        date + "T12:00:00"
                                    ).getDay() === 0
                                ) {

                                    status = "SUNDAY / WEEKLY OFF";

                                }

                                else if (
                                    first !== null &&
                                    first < 12 * 60 &&
                                    last !== null &&
                                    last < HALF_DAY_CUTOFF
                                ) {

                                    status = "HALF DAY - MORNING";

                                }

                                else if (
                                    first !== null &&
                                    first >= HALF_DAY_CUTOFF
                                ) {

                                    status = "HALF DAY - AFTERNOON";

                                }


                                let afternoonLogoutMin = null;
                                let afternoonLoginMin = null;

                                if (logouts.length >= 2) {
                                    afternoonLogoutMin = logouts[0];
                                } else {
                                    afternoonLogoutMin = logouts.find(m => m >= 13 * 60 && m <= 15 * 60) || null;
                                }

                                if (logins.length >= 2) {
                                    afternoonLoginMin = logins[1];
                                } else {
                                    afternoonLoginMin = logins.find(m => m >= 14 * 60 && m <= 16 * 60) || null;
                                }

                                let morningLate = 0;
                                if (
                                    first !== null &&
                                    first > WORK_START &&
                                    status !== "SUNDAY / WEEKLY OFF" &&
                                    status !== "ABSENT"
                                ) {
                                    morningLate = first - WORK_START;
                                }

                                let afternoonLate = 0;
                                if (
                                    afternoonLoginMin !== null &&
                                    afternoonLoginMin > LUNCH_END &&
                                    status !== "SUNDAY / WEEKLY OFF" &&
                                    status !== "ABSENT"
                                ) {
                                    afternoonLate = afternoonLoginMin - LUNCH_END;
                                }

                                const totalLate = morningLate + afternoonLate;

                                return {

                                    _index: i + 2,

                                    date: date,

                                    day: getDayName(date),

                                    employeeId: id,

                                    employeeName: name || id,

                                    status: status,

                                    attendanceValue:
                                        status === "FULL DAY PRESENT"
                                            ? 1
                                            : status.includes("HALF DAY")
                                                ? .5
                                                : 0,

                                    lateMinutes: totalLate,

                                    morningLateMinutes: morningLate,

                                    afternoonLateMinutes: afternoonLate,

                                    lateFine: 0,

                                    firstLogin:
                                        minutesToTime(first),

                                    afternoonLogout:
                                        minutesToTime(afternoonLogoutMin),

                                    afternoonLogin:
                                        minutesToTime(afternoonLoginMin),

                                    lastLogout:
                                        minutesToTime(last),

                                    punchCount:
                                        logins.length +
                                        logouts.length

                                };

                            })
                            .filter(
                                x => (x.employeeId || x.employeeName) &&
                                    x.date
                            );


                    filteredData = [...rawData];


                    localStorage.setItem(
                        "attendanceRawTXTDataV3",
                        JSON.stringify(rawData)
                    );


                    fileInfo.textContent =
                        `Loaded Excel: ${file.name} • ${rawData.length} records`;


                    sideStatus.textContent =
                        `${file.name.slice(0, 22).toUpperCase()} LOADED`;


                    populateEmployees();

                    resetFilters();

                    renderAll();

                    showPage("dashboard");


                } catch (err) {

                    console.error(err);

                    alert(
                        "Could not read this Excel file."
                    );

                }

            };

            reader.readAsArrayBuffer(file);

        }

        function populateEmployees() {

            const current =
                employeeFilter.value;


            const map =
                new Map();


            rawData.forEach(x => {

                const id =
                    x.employeeId ||
                    x.employeeName;


                if (!map.has(id)) {

                    map.set(
                        id,
                        {
                            id: id,
                            name:
                                x.employeeName ||
                                id
                        }
                    );

                }

            });


            const employees =
                [...map.values()]
                    .sort(
                        (a, b) =>
                            String(a.name)
                                .localeCompare(
                                    String(b.name)
                                )
                    );

                '<option value="">All Employees</option>' +

                employees

                        e =>
                            `<option value="${escapeHtml(e.id)}">
     ${escapeHtml(e.name)}

    </option>`
                    )

        }


        /* =========================================================
           FILTERS
        ========================================================= */

        function applyFilters() {

            const q =
                searchInput.value
                    .toLowerCase()
                    .trim();


            const date =
                dateFilter.value;


            const status =
                statusFilter.value;


            const emp =
                employeeFilter.value;


            filteredData =
                rawData.filter(x =>

                    (
                        !q ||
                        `${x.employeeId} ${x.employeeName}`
                            .toLowerCase()
                            .includes(q)
                    )

                    &&

                    (
                        !date ||
                        x.date === date
                    )

                    &&

                    (
                        !status ||
                        x.status === status
                    )

                    &&

                    (
                        !emp ||
                        x.employeeId === emp
                    )

                );


            renderAll();

        }


        function resetFilters() {

            searchInput.value = "";
            dateFilter.value = "";
            statusFilter.value = "";
            employeeFilter.value = "";

            filteredData = [...rawData];

            renderAll();

        }


        /* =========================================================
           BADGES
        ========================================================= */

        function badge(status) {

            let cls = "b-gray";

            if (status === "FULL DAY PRESENT") {
                cls = "b-green";
            }

            else if (
                status.includes("MORNING")
            ) {

                cls = "b-amber";

            }

            else if (
                status.includes("AFTERNOON")
            ) {

                cls = "b-purple";

            }

            else if (
                status === "ABSENT"
            ) {

                cls = "b-red";

            }

            return `
 <span class="badge ${cls}">
 ${escapeHtml(status)}
 </span>
 `;

        }


        /* =========================================================
           STATISTICS
        ========================================================= */

        function stats(data = filteredData) {

            const count =
                status =>
                    data.filter(
                        x => x.status === status
                    ).length;


            return {

                emp:
                    new Set(
                        data.map(
                            x => x.employeeId || x.employeeName
                        )
                    ).size,

                full:
                    count("FULL DAY PRESENT"),

                morning:
                    count("HALF DAY - MORNING"),

                afternoon:
                    count("HALF DAY - AFTERNOON"),

                absent:
                    count("ABSENT"),

                late:
                    data.reduce(
                        (a, x) =>
                            a +
                            Number(x.lateMinutes || 0),
                        0
                    )

            };

        }

        /* =========================================================
           EMPLOYEE SUMMARY
        ========================================================= */

        function employeeSummary() {

            const map = new Map();


            filteredData.forEach(x => {

                const key =
                    x.employeeId ||
                    x.employeeName;


                if (!map.has(key)) {

                    map.set(
                        key,
                        {
                            id: x.employeeId,
                            name: x.employeeName,
                            full: 0,
                            morning: 0,
                            afternoon: 0,
                            absent: 0,
                            weeklyOff: 0,
                            leaveDates: new Set(),
                            lateMinutes: 0
                        }
                    );

                }


                const a = map.get(key);


                if (
                    x.status === "FULL DAY PRESENT"
                ) {

                    a.full++;

                }


                if (
                    x.status === "HALF DAY - MORNING"
                ) {

                    a.morning++;

                    if (x.date) {
                        a.leaveDates.add(x.date);
                    }

                }


                if (
                    x.status === "HALF DAY - AFTERNOON"
                ) {

                    a.afternoon++;

                    if (x.date) {
                        a.leaveDates.add(x.date);
                    }

                }


                if (
                    x.status === "ABSENT"
                ) {

                    a.absent++;

                    if (x.date) {
                        a.leaveDates.add(x.date);
                    }

                }


                if (
                    x.status === "SUNDAY / WEEKLY OFF"
                ) {

                    a.weeklyOff++;

                }


                a.lateMinutes +=
                    Number(x.lateMinutes || 0);

            });


            return [...map.values()]
                .map(x => {

                    const dates =
                        [...x.leaveDates]
                            .sort();


                    return {

                        ...x,

                        leaveCount:
                            dates.length,

                        leaveDates:
                            dates,

                        presentEquivalent:
                            x.full +
                            (
                                x.morning +
                                x.afternoon
                            ) * .5

                    };

                });

        }


        /* =========================================================
           EMPLOYEE REPORT
        ========================================================= */

        function renderEmployeeReport() {

            const q =
                (
                    employeeReportSearch.value ||
                    ""
                )
                    .toLowerCase();


            const sort =
                employeeSort.value;


            let data =
                employeeSummary()
                    .filter(
                        x =>
                            `${x.id} ${x.name}`
                                .toLowerCase()
                                .includes(q)
                    );


            data.sort(
                (a, b) => {

                    if (sort === "present") {
                        return b.presentEquivalent -
                            a.presentEquivalent;
                    }

                    if (sort === "late") {
                        return b.lateMinutes -
                            a.lateMinutes;
                    }

                    return String(a.name)
                        .localeCompare(
                            String(b.name)
                        );

                }
            );


            if (!data.length) {

                employeeRows.innerHTML =
                    `
   <tr>
    <td colspan="10" class="empty">
     No data to display.
    </td>
   </tr>
   `;

                return;

            }


            employeeRows.innerHTML =
                data.map(x => {

                    const leaveDates =
                        x.leaveDates
                            .map(
                                d => formatDate(d)
                            )
                            .join(", ") || "—";


                    return `
   <tr>

    <td>
     ${escapeHtml(x.id)}
    </td>

    <td>
     <b>
      ${escapeHtml(x.name)}
     </b>
    </td>

    <td>
     ${x.full}
    </td>

    <td>
     ${x.morning}
    </td>

    <td>
     ${x.afternoon}
    </td>

    <td>
     ${x.absent}
    </td>

    <td>
     ${x.presentEquivalent}
    </td>

    <td>
     ${x.leaveCount}
    </td>

    <td style="white-space:normal;min-width:200px">
     ${escapeHtml(leaveDates)}
    </td>

    <td>
     ${x.lateMinutes}
    </td>

   </tr>
   `;

                })
                    .join("");

        }


        /* =========================================================
           DATE REPORT
        ========================================================= */

        function renderDates() {

            const map = new Map();


            filteredData.forEach(x => {

                if (!x.date) {
                    return;
                }


                if (!map.has(x.date)) {

                    map.set(
                        x.date,
                        {
                            day: x.day,
                            full: 0,
                            morning: 0,
                            afternoon: 0,
                            absent: 0,
                            off: 0,
                            lateEmp: new Set(),
                            lateMinutes: 0
                        }
                    );

                }


                const a = map.get(x.date);


                if (
                    x.status === "FULL DAY PRESENT"
                ) {
                    a.full++;
                }


                if (
                    x.status === "HALF DAY - MORNING"
                ) {
                    a.morning++;
                }


                if (
                    x.status === "HALF DAY - AFTERNOON"
                ) {
                    a.afternoon++;
                }


                if (
                    x.status === "ABSENT"
                ) {
                    a.absent++;
                }


                if (
                    x.status === "SUNDAY / WEEKLY OFF"
                ) {
                    a.off++;
                }


                if (
                    Number(x.lateMinutes) > 0
                ) {

                    a.lateEmp.add(
                        x.employeeId ||
                        x.employeeName
                    );

                    a.lateMinutes +=
                        Number(x.lateMinutes);

                }

            });


            const arr =
                [...map.entries()]
                    .sort(
                        (a, b) =>
                            a[0].localeCompare(b[0])
                    );


            if (!arr.length) {

                dateRows.innerHTML =
                    `
   <tr>
    <td colspan="9" class="empty">
     No data to display.
    </td>
   </tr>
   `;

                return;

            }


            dateRows.innerHTML =
                arr.map(
                    ([date, x]) =>
                        `
   <tr>

    <td>
     ${formatDate(date)}
    </td>

    <td>
     ${escapeHtml(x.day)}
    </td>

    <td>
     ${x.full}
    </td>

    <td>
     ${x.morning}
    </td>

    <td>
     ${x.afternoon}
    </td>

    <td>
     ${x.absent}
    </td>

    <td>
     ${x.off}
    </td>

    <td>
     ${x.lateEmp.size}
    </td>

    <td>
     ${x.lateMinutes}
    </td>

   </tr>
   `
                )
                    .join("");

        }

        function exportFilteredCSV() {

            if (!filteredData.length) {

                alert(
                    "No processed data available to export."
                );

                return;

            }


            const headers = [

                "Date",
                "Day",
                "Employee ID",
                "Employee Name",
                "Attendance Status",
                "Attendance Value",
                "First Login",
                "Afternoon Check-out",
                "Afternoon Check-in",
                "Last Logout",
                "Punch Count",
                "Punch Times",
                "Morning Late Minutes",
                "Afternoon Late Minutes",
                "Total Late Minutes"

            ];


            const rows = [
                headers,

                ...filteredData.map(
                    x => [

                        formatDate(x.date),

                        x.day,

                        x.employeeId,

                        x.employeeName,

                        x.status,

                        x.attendanceValue,

                        x.firstLogin,

                        x.afternoonLogout || "",

                        x.afternoonLogin || "",

                        x.lastLogout,

                        x.punchCount,

                        x.punchTimes,

                        x.morningLateMinutes || 0,

                        x.afternoonLateMinutes || 0,

                        x.lateMinutes || 0

                    ]
                )

            ];


            const csv =
                rows
                    .map(
                        row =>
                            row
                                .map(
                                    value =>
                                        `"${String(value ?? "")
                                            .replaceAll('"', '""')}"`
                                )
                                .join(",")
                    )
                    .join("\n");


            const blob =
                new Blob(
                    [csv],
                    { type: "text/csv;charset=utf-8;" }
                );


            const url =
                URL.createObjectURL(blob);


            const a =
                document.createElement("a");


            a.href = url;

            a.download =
                "processed_attendance_data.csv";


            document.body.appendChild(a);

            a.click();

            document.body.removeChild(a);

            URL.revokeObjectURL(url);

        }

                    ) || "[]"
                );

                Array.isArray(saved) &&
                saved.length
            ) {

            );

        ========================================================= */

        renderAll();
