/* =========================================================

const WORK_START =
    9 * 60 + 30;                 // 09:30 AM

const LATE_GRACE_TIME =
    9 * 60 + 35;                 // 09:35 AM

const AFTERNOON_CHECKOUT =
    13 * 60 + 55;                     // 01:55 PM

const AFTERNOON_CHECKIN =
    14 * 60 + 30;                // 02:30 PM

const MORNING_HALF_DAY_LIMIT =
    10 * 60 + 30;                // 10:30 AM

/* =========================================================
   BASIC HELPERS
========================================================= */
function escapeHtml(v){
    return String(v ?? "").replace(
        /[&<>"']/g,
        function(m){
            return {
                "&":"&amp;",
                "<":"&lt;",
                ">":"&gt;",
                '"':"&quot;",
                "'":"&#039;"
            }[m];

        }
    );
}

function norm(s){
    return String(s || "")
        .trim()
        .toLowerCase()
        .replace(/[^a-z0-9]/g,"");

}
function minutesToTime(m){

    if(
        m === null ||
        m === undefined ||
        Number.isNaN(m)
    ){
        return "";
    }

    return String(
        Math.floor(m / 60)
    ).padStart(2,"0")
    + ":"
    + String(
        m % 60
    ).padStart(2,"0");

}


function formatDate(date){

    if(!date){
        return "";
    }

    const p = date.split("-");

    if(p.length !== 3){
        return date;
    }

    const months = [
        "JAN","FEB","MAR","APR","MAY","JUN",
        "JUL","AUG","SEP","OCT","NOV","DEC"
    ];

    return p[2]
        + "-"
        + months[Number(p[1])-1]
        + "-"
        + p[0];

}
function getDayName(date){

    if(!date){
        return "";
    }

    const d =
        new Date(date + "T12:00:00");

    if(isNaN(d)){
        return "";
    }

    return d.toLocaleDateString(
        "en-US",
        {
            weekday:"long"
        }
    );

}


function getTimeFromTimestamp(timestamp){

    const match =
        String(timestamp).match(
            /(\d{1,2}):(\d{2})(?::(\d{2}))?/
        );

    if(!match){
        return null;
    }
    return Number(match[1]) * 60
        + Number(match[2]);
}
function parseDateValue(v){

    const s =
        String(v || "").trim();

    if(!s){
        return "";
    }

    const iso =
        s.match(
            /^(\d{4})-(\d{1,2})-(\d{1,2})$/
        );

    if(iso){

        return iso[1]
            + "-"
            + String(iso[2]).padStart(2,"0")
            + "-"
            + String(iso[3]).padStart(2,"0");

    }

    const dt =
        new Date(s);

    if(!isNaN(dt)){

        return dt.getFullYear()
            + "-"
            + String(
                dt.getMonth()+1
            ).padStart(2,"0")
            + "-"
            + String(
                dt.getDate()
            ).padStart(2,"0");

    }
    return "";
}
/* =========================================================
   PUNCH ANALYSIS
========================================================= */
function analyzePunches(list){

    const sorted =
        [...list].sort(
            function(a,b){
                return a.minutes - b.minutes;
            }
        );


    const firstMinutes =
        sorted.length
        ? sorted[0].minutes
        : null;


    const lastMinutes =
        sorted.length
        ? sorted[sorted.length - 1].minutes
        : null;


    /*
       ALL PUNCHES AFTER 01:55 PM
    */

    const afternoonPunches =
        sorted.filter(
            function(p){

                return (
                    p.minutes >= AFTERNOON_CHECKOUT
                );

            }
        );


    /*
       FIRST PUNCH AFTER 01:55 PM
       = AFTERNOON CHECK-OUT
    */

    const afternoonCheckout =
        afternoonPunches.length > 0
        ? afternoonPunches[0].minutes
        : null;


    /*
       SECOND PUNCH AFTER 01:55 PM
       = AFTERNOON CHECK-IN
    */

    const afternoonCheckin =
        afternoonPunches.length > 1
        ? afternoonPunches[1].minutes
        : null;


    return {

        firstMinutes:
            firstMinutes,

        lastMinutes:
            lastMinutes,

        afternoonCheckout:
            afternoonCheckout,

        afternoonCheckin:
            afternoonCheckin

    };

}

/* =========================================================
   ATTENDANCE CALCULATION
========================================================= */

function calculateAttendance(
    firstMinutes,
    lastMinutes,
    date,
    afternoonCheckout = null,
    afternoonCheckin = null
){
    let status = "ABSENT";
    /*
       NO PUNCH
    */

    if(
        firstMinutes === null &&
        lastMinutes === null
    ){
        status = "ABSENT";
    }
     else if(firstMinutes > MORNING_HALF_DAY_LIMIT){      //morning half day limits
        status = "ABSENT IN THE MORNING";
     }
    /*
       MORNING HALF DAY
       First punch before 2 PM
       AND last punch at/before 2 PM.
    */

    else if(
        firstMinutes < AFTERNOON_CHECKOUT &&
        lastMinutes <= AFTERNOON_CHECKOUT
    ){

        status =
            "PRESENT IN THE MORNING";

    }

    /*
       AFTERNOON HALF DAY
       First punch at or after
       2:30 PM.
    */
    else if(
        firstMinutes >= AFTERNOON_CHECKIN
    ){
        status =
            "PRESENT IN THE AFTERNOON";
    }
    /*
       FULL DAY
       Morning attendance + afternoon
       attendance.
    */

    else if(
        firstMinutes < AFTERNOON_CHECKOUT &&
        lastMinutes > AFTERNOON_CHECKIN
    ){

        status =
            "FULL DAY PRESENT";

    }

    else{

        status =
            "FULL DAY PRESENT";

    }
    /*
       SUNDAY / WEEKLY OFF
    */
    if(date){

        const dt =
            new Date(
                date + "T12:00:00"
            );

        if(
            dt.getDay() === 0 &&
            status === "ABSENT"
        ){

            status =
                "SUNDAY / WEEKLY OFF";

        }

    }

    /*
       MORNING LATE
    */
    let morningLateMinutes = 0;

    if(
        firstMinutes !== null &&
        firstMinutes > LATE_GRACE_TIME &&
        status !== "SUNDAY / WEEKLY OFF"
    ){
        morningLateMinutes =
            firstMinutes -
            WORK_START;
    }
    /*
       AFTERNOON LATE
    */
    let afternoonLateMinutes = 0;
    if(
        afternoonCheckin !== null &&
        afternoonCheckin >
            AFTERNOON_CHECKIN &&
        status !== "SUNDAY / WEEKLY OFF"
    ){

        afternoonLateMinutes =
            afternoonCheckin -
            AFTERNOON_CHECKIN;

    }
    const totalLateMinutes =
        morningLateMinutes +
        afternoonLateMinutes;

      let lateCount = 0;

if(morningLateMinutes > 0){
    lateCount++;
}

if(afternoonLateMinutes > 0){
    lateCount++;
}

    return {

        status:status,


        morningLateMinutes:
            morningLateMinutes,

        afternoonLateMinutes:
            afternoonLateMinutes,

        lateMinutes:
            totalLateMinutes,
          lateCount:lateCount

    };

}

/* =========================================================
   RAW TXT PARSER
========================================================= */

function parseRawTxt(text){

    const lines =
        text
        .replace(/\r\n/g,"\n")
        .replace(/\r/g,"\n")
        .split("\n");
    rawLines = lines;

    const punches = [];

    lines.forEach(
        function(line,index){

            const clean =
                line.trim();

            if(!clean){
                return;
            }
            /*
               Expected:

               378  2026-08-31 09:27:45  1 0 3 0
            */
            const match =
                clean.match(
                    /^\s*(\d+)\s+(\d{4}-\d{2}-\d{2})\s+(\d{1,2}:\d{2}(?::\d{2})?)\s*(.*)$/
                );

            if(!match){
                return;
            }
            const employeeId =
                match[1];
            const date =
                match[2];
            const time =
                match[3];
            const minutes =
                getTimeFromTimestamp(time);
            if(minutes === null){
                return;
            }
            punches.push({
                employeeId:employeeId,
                date:date,
                time:time,
                minutes:minutes,
                line:index + 1
            });
        }
    );
    const grouped =
        new Map();
    punches.forEach(
        function(p){

            const key =
                p.employeeId +
                "|" +
                p.date;


            if(!grouped.has(key)){

                grouped.set(
                    key,
                    []
                );
            }

            grouped
                .get(key)
                .push(p);

        }
    );

    rawData =
        Array.from(
            grouped.entries()
        )
        .map(
            function(entry,index){
                const key = entry[0];
                const list = entry[1];

                list.sort(
                    function(a,b){

                        return (
                            a.minutes -
                            b.minutes
                        );
                    }
                );
                const parts =
                    key.split("|");

                const employeeId =
                    parts[0];
                const date =
                    parts[1];
                const analysis =
                    analyzePunches(list);

                const result =
                    calculateAttendance(
                        analysis.firstMinutes,
                        analysis.lastMinutes,
                        date,
                        analysis.afternoonCheckout,
                        analysis.afternoonCheckin
                    );

                return {
    _index:index + 1,

    date:date,

    day:getDayName(date),

    employeeId:employeeId,

    employeeName:"",

    status:result.status,
    firstLogin:minutesToTime(
            analysis.firstMinutes
        ),

    afternoonCheckout:
        analysis.afternoonCheckout !== null
            ? minutesToTime(
                analysis.afternoonCheckout
            )
            : "",

    afternoonCheckin:
        analysis.afternoonCheckin !== null
            ? minutesToTime(
                analysis.afternoonCheckin
            )
            : "",

    lastLogout:
        analysis.lastMinutes !== null
            ? minutesToTime(
                analysis.lastMinutes
            )
            : "",

    /* MORNING LATE MINUTES */

    morningLateMinutes:
        result.morningLateMinutes,

    /* AFTERNOON LATE MINUTES */

    afternoonLateMinutes:
        result.afternoonLateMinutes,

    /* TOTAL LATE MINUTES */

    lateMinutes:
        result.lateMinutes,


    /* LATE COUNT: 0, 1 OR 2 */

    lateCount:
        result.lateCount,

    lateFine:
        0,

    punchTimes:
        list.map(
            function(x){
                return x.time;
            }
        ).join(", ")

};
      }
        )
        .sort(
            function(a,b){

                return (
                    a.date.localeCompare(
                        b.date
                    )
                    ||
                    a.employeeId.localeCompare(
                        b.employeeId
                    )
                );

            }
        );

    filteredData =
        [...rawData];

    localStorage.setItem(
        "attendanceRawTXTDataV4",
        JSON.stringify(rawData)
    );

    rawPreview.textContent =
        lines
        .slice(0,500)
        .join("\n")
        ||
        "No valid raw data found.";

    rawInfo.textContent =
        punches.length
        + " raw punches found • "
        + rawData.length
        + " employee-date attendance records created.";

    return rawData;
}
/* =========================================================
   CSV PARSER
========================================================= */
function csvParse(text){
    const rows = [];
    let row = [];
    let cell = "";
    let q = false;
    for(
        let i=0;
        i<text.length;
        i++
    ){
        const c =
            text[i];

        const n =
            text[i+1];
        if(c === '"'){
            if(
                q &&
                n === '"'
            ){
                cell += '"';
                i++;
            }
            else{
                q = !q;

            }

        }

        else if(
            c === "," &&
            !q
        ){

            row.push(cell);

            cell = "";

        }

        else if(
            (
                c === "\n" ||
                c === "\r"
            ) &&
            !q
        ){

            if(
                c === "\r" &&
                n === "\n"
            ){

                i++;

            }

            row.push(cell);

            if(
                row.some(
                    function(x){
                        return (
                            x.trim() !== ""
                        );
                    }
                )
            ){

                rows.push(row);

            }
            row = [];
            cell = "";
        }
        else{
            cell += c;
        }
    }

    if(
        cell.length ||
        row.length
    ){
        row.push(cell);
        rows.push(row);
    }
    return rows;
}

function pick(obj,...names){
    for(
        const n of names
    ){
        if(
            obj[n] !== undefined &&
            obj[n] !== ""
        )
        {
            return obj[n];
        }
    }

    return "";
}

function canonicalStatus(v){

    const s =
        String(v || "")
        .trim()
        .toUpperCase();


    if(
        s.includes(
            "PRESENT IN THE MORNING"
        ) ||
        s.includes("MORNING")
    ){

        return "PRESENT IN THE MORNING";

    }

    if(
        s.includes(
            "PRESENT IN THE AFTERNOON"
        ) ||
        s.includes("AFTERNOON") ||
        s.includes("NOON")
    ){

        return "PRESENT IN THE AFTERNOON";

    }

    if(
        s.includes("FULL") ||
        s === "P" ||
        s === "PRESENT"
    ){

        return "FULL DAY PRESENT";

    }

    if(
        s.includes("ABSENT") ||
        s === "A"
    ){
        return "ABSENT";
    }

    if(
        s.includes("SUNDAY") ||
        s.includes("WEEKLY OFF") ||
        s === "OFF"
    ){
        return "SUNDAY / WEEKLY OFF";

    }
    return s || "UNKNOWN";
}

/* =========================================================
   EXCEL DATE
========================================================= */

function excelDate(v){

    if(
        v === null ||
        v === undefined ||
        String(v).trim() === ""
    ){

        return "";

    }


    if(
        v instanceof Date &&
        !isNaN(v.getTime())
    ){

        return v.getFullYear()
            + "-"
            + String(
                v.getMonth()+1
            ).padStart(2,"0")
            + "-"
            + String(
                v.getDate()
            ).padStart(2,"0");

    }


    if(typeof v === "number"){

        if(
            !Number.isFinite(v) ||
            v < 1
        ){

            return "";

        }


        const p =
            XLSX.SSF.parse_date_code(
                Math.floor(v)
            );


        if(
            p &&
            p.y >= 2000 &&
            p.y <= 2100
        ){

            return p.y
                + "-"
                + String(p.m).padStart(2,"0")
                + "-"
                + String(p.d).padStart(2,"0");

        }

    }


    return parseDateValue(v);

}


/* =========================================================
   TIME PARSER FOR EXCEL
========================================================= */

function timeToMinutes(v){

    if(
        v === null ||
        v === undefined ||
        v === ""
    ){

        return null;

    }


    if(v instanceof Date){

        return (
            v.getHours() * 60 +
            v.getMinutes()
        );

    }


    if(typeof v === "number"){

        if(
            v >= 0 &&
            v < 1
        ){

            return Math.round(
                v * 24 * 60
            );

        }


        const p =
            XLSX.SSF.parse_date_code(v);


        if(p){

            return (
                p.H * 60 +
                p.M
            );

        }

    }


    return getTimeFromTimestamp(v);

}

   EXCEL SHEET

function sheetRowsToObjects(ws){

    const rows =
        XLSX.utils.sheet_to_json(
            ws,
            {
                header:1,
                defval:"",
                raw:true
            }
        );


    if(!rows.length){
        return [];
    }


    const headers =
        rows[0].map(
            function(h,i){

                return (
                    norm(h) ||
                    "col" + i
                );

            }
        );


    return rows
        .slice(1)
        .map(
            function(arr){

                const o = {};


                headers.forEach(
                    function(h,i){

                        o[h] =
                            arr[i] ?? "";

                    }
                );


                return o;

            }
        );

}

   EXCEL LOAD
========================================================= */

function loadExcel(file){

    const reader =
        new FileReader();


    reader.onload =
        function(e){

            try{

                const wb =
                    XLSX.read(
                        new Uint8Array(
                            e.target.result
                        ),
                        {
                            type:"array",
                            cellDates:true
                        }
                    );


                const punchName =
                    wb.SheetNames.find(
                        n =>
                            norm(n) ===
                            "punchingtime"
                    )
                    ||
                    wb.SheetNames.find(
                        n =>
                            norm(n)
                            .includes("punch")
                    );


                if(!punchName){

                    alert(
                        "Could not find the Punching Time sheet."
                    );

                    return;

                }


                const punch =
                    sheetRowsToObjects(
                        wb.Sheets[punchName]
                    );


                rawData =
                    punch.map(
                        function(o,index){

                            const employeeId =
                                String(
                                    pick(
                                        o,
                                        "empid",
                                        "employeeid",
                                        "id"
                                    )
                                ).trim();


                            const employeeName =
                                String(
                                    pick(
                                        o,
                                        "name",
                                        "nameofemployee",
                                        "employeename"
                                    )
                                ).trim();


                            const date =
                                excelDate(
                                    pick(
                                        o,
                                        "date"
                                    )
                                );


                            const logins = [];

                            const logouts = [];


                            for(
                                let n=1;
                                n<=10;
                                n++
                            ){

                                const li =
                                    n === 1
                                    ? o["login"]
                                    : o[
                                        "login" + n
                                    ];


                                const lo =
                                    n === 1
                                    ? o["logout"]
                                    : o[
                                        "logout" + n
                                    ];


                                const a =
                                    timeToMinutes(li);


                                const b =
                                    timeToMinutes(lo);


                                if(a !== null){

                                    logins.push(a);

                                }


                                if(b !== null){

                                    logouts.push(b);

                                }

                            }


                            const first =
                                logins.length
                                ? Math.min(
                                    ...logins
                                )
                                : null;


                            const last =
                                logouts.length
                                ? Math.max(
                                    ...logouts
                                )
                                : null;


                            /*
                               ==========================================
                               AFTERNOON CHECK-OUT
                               ==========================================

                               EXACT MATCH ONLY: only a punch at
                               exactly 02:00 PM counts.
                            */

                            const allPunches =
                                [
                                    ...logins,
                                    ...logouts
                                ];


                            const afternoonCheckout =
                                allPunches.includes(
                                    AFTERNOON_CHECKOUT
                                )
                                ? AFTERNOON_CHECKOUT
                                : null;


                            /*
                               ==========================================
                               AFTERNOON CHECK-IN
                               ==========================================

                               EXACT MATCH ONLY: only a punch at
                               exactly 02:30 PM counts.
                            */

                            const afternoonCheckin =
                                allPunches.includes(
                                    AFTERNOON_CHECKIN
                                )
                                ? AFTERNOON_CHECKIN
                                : null;


                            const result =
                                calculateAttendance(
                                    first,
                                    last,
                                    date,
                                    afternoonCheckout,
                                    afternoonCheckin
                                );


                            return {

                                _index:
                                    index + 2,

                                date:
                                    date,

                                day:
                                    getDayName(date),

                                employeeId:
                                    employeeId,

                                employeeName:
                                    employeeName,

                                status:
                                    result.status,

                                attendanceValue:
                                    result.attendanceValue,

                                firstLogin:
                                    minutesToTime(first),

                                afternoonCheckout:
                                    minutesToTime(
                                        afternoonCheckout
                                    ),

                                afternoonCheckin:
                                    minutesToTime(
                                        afternoonCheckin
                                    ),

                                lastLogout:
                                    minutesToTime(last),

                                punchCount:
                                    logins.length +
                                    logouts.length,

                                morningLateMinutes:
                                    result.morningLateMinutes,

                                afternoonLateMinutes:
                                    result.afternoonLateMinutes,

                                lateMinutes:
                                    result.lateMinutes,

                                lateFine:
                                    0

                            };

                        }
                    )
                    .filter(
                        function(x){

                            return (
                                (
                                    x.employeeId ||
                                    x.employeeName
                                )
                                &&
                                x.date
                            );

                        }
                    );


                filteredData =
                    [...rawData];


                localStorage.setItem(
                    "attendanceRawTXTDataV4",
                    JSON.stringify(rawData)
                );


                fileInfo.textContent =
                    "Loaded Excel: "
                    + file.name
                    + " • "
                    + rawData.length
                    + " records";


                sideStatus.textContent =
                    file.name
                    .slice(0,22)
                    .toUpperCase()
                    + " LOADED";


                populateEmployees();

                resetFilters();

                renderAll();

                showPage("dashboard");

            }

            catch(err){

                console.error(err);

                alert(
                    "Could not read this Excel file. Please check its format."
                );

            }

        };


    reader.readAsArrayBuffer(file);

}

function handleFile(file){

    const name =
        file.name.toLowerCase();


    if(name.endsWith(".txt")){

        const reader =
            new FileReader();


        reader.onload =
            function(e){

                parseRawTxt(
                    e.target.result
                );


                fileInfo.textContent =
                    "Loaded RAW TXT: "
                    + file.name
                    + " • "
                    + rawData.length
                    + " attendance records";


                sideStatus.textContent =
                    file.name
                    .slice(0,22)
                    .toUpperCase()
                    + " LOADED";


                populateEmployees();

                resetFilters();

                renderAll();

                showPage("dashboard");

            };


        reader.readAsText(file);

    }


    else if(name.endsWith(".csv")){

        loadCSV(file);

    }


    else if(
        name.endsWith(".xlsx") ||
        name.endsWith(".xls")
    ){

        loadExcel(file);

    }


    else{

        alert(
            "Please upload TXT, CSV, XLSX or XLS."
        );

    }

}

function populateEmployees(){

    const current =
        employeeFilter.value;


    const employees =
        [
            ...new Map(
                rawData.map(
                    function(x){

                        const key =
                            x.employeeId ||
                            x.employeeName;


                        return [
                            key,
                            {
                                id:
                                    x.employeeId,

                                name:
                                    x.employeeName ||
                                    x.employeeId
                            }
                        ];

                    }
                )
            ).values()
        ]
        .sort(
            function(a,b){

                return a.name.localeCompare(
                    b.name
                );

            }
        );


    employeeFilter.innerHTML =
        '<option value="">All Employees</option>'
        +
        employees.map(
            function(e){

                return (
                    '<option value="' +
                    escapeHtml(e.id) +
                    '">' +
                    escapeHtml(e.name) +
                    ' (' +
                    escapeHtml(e.id) +
                    ')</option>'
                );

            }
        ).join("");


    employeeFilter.value =
        current;

}


function applyFilters(){

    const q =
        searchInput.value.toLowerCase();


    const date =
        dateFilter.value;


    const status =
        statusFilter.value;


    const emp =
        employeeFilter.value;


    filteredData =
        rawData.filter(
            function(x){

                return (

                    (
                        !q ||
                        String(
                            x.employeeId
                            + " "
                            + x.employeeName
                        )
                        .toLowerCase()
                        .includes(q)
                    )

                    &&

                    (
                        !date ||
                        x.date === date
                    )

                    &&

                    (
                        !status ||
                        x.status === status
                    )

                    &&

                    (
                        !emp ||
                        x.employeeId === emp
                    )

                );

            }
        );


    renderAll();

}


function resetFilters(){

    searchInput.value = "";

    dateFilter.value = "";

    statusFilter.value = "";

    employeeFilter.value = "";

    filteredData =
        [...rawData];


    renderAll();

}


/* =========================================================
   BADGE
========================================================= */

function badge(status){

    let cls =
        "b-gray";


    if(
        status ===
        "FULL DAY PRESENT"
    ){

        cls =
            "b-green";

    }

    else if(
        status ===
        "PRESENT IN THE MORNING"
    ){

        cls =
            "b-amber";

    }

    else if(
        status ===
        "PRESENT IN THE AFTERNOON"
    ){

        cls =
            "b-purple";

    }

    else if(
        status ===
        "ABSENT"
    ){

        cls =
            "b-red";

    }


    return (
        '<span class="badge '
        + cls
        + '">'
        + escapeHtml(status)
        + '</span>'
    );

}


/* =========================================================
   STATS
========================================================= */

function stats(
    data = filteredData
){

    const count =
        function(status){

            return data.filter(
                x =>
                    x.status === status
            ).length;

        };


    return {

        emp:
            new Set(
                data.map(
                    x =>
                        x.employeeId ||
                        x.employeeName
                )
            ).size,

        full:
            count(
                "FULL DAY PRESENT"
            ),

        morning:
            count(
                "PRESENT IN THE MORNING"
            ),

        afternoon:
            count(
                "PRESENT IN THE AFTERNOON"
            ),

        absent:
            count(
                "ABSENT"
            ),

        morningLate:
            data.reduce(
                (
                    sum,x
                ) =>
                    sum +
                    (
                        x.morningLateMinutes ||
                        0
                    ),
                0
            ),

        afternoonLate:
            data.reduce(
                (
                    sum,x
                ) =>
                    sum +
                    (
                        x.afternoonLateMinutes ||
                        0
                    ),
                0
            ),

        late:
            data.reduce(
                (
                    sum,x
                ) =>
                    sum +
                    (
                        x.lateMinutes ||
                        0
                    ),
                0
            )

    };

}

    }

                        "Present Morning",
                        "Present Afternoon",
                        "Absent"
                    ],

                    datasets:[{

                        data:[
                            s.full,
                            s.morning,
                            s.afternoon,
                            s.absent
                        ],

                        backgroundColor:[
                            "#1fa774",
                            "#e99a22",
                            "#7654d6",
                            "#dc4c64"
                        ]

                    }]

                },

                options:{
                    responsive:true,
                    maintainAspectRatio:false
                }

            }
        );

       LATE BY EMPLOYEE
    */

        new Map();

        function(x){

                x.employeeId ||

            if(!lateMap.has(key)){

                lateMap.set(
                    key,
                    {
                        name:
                            x.employeeName ||
                            x.employeeId,

                        mins:0
                    }
                );

            }


            lateMap.get(key).mins +=
                x.lateMinutes || 0;

        }
    );


    const lateEmployees =
        [
            ...lateMap.values()
        ]
        .sort(
            function(a,b){

                return (
                    b.mins -
                    a.mins
                );

            }
        );


    charts.late =
        new Chart(
            lateChart,
            {

                type:"bar",

                data:{

                    labels:
                        lateEmployees.map(
                            x =>
                                x.name

                    datasets:[{

                        label:

                    maintainAspectRatio:false,

                        }

        );

    filteredData.forEach(
        function(x){

            if(!x.date){
                return;
            }


            if(!byDate[x.date]){
                byDate[x.date] = 0;
            }


            if(
                x.status ===
                "FULL DAY PRESENT"
            ){

                byDate[x.date]++;

            }

        }
    );


    const dates =
        Object.keys(
            byDate
        ).sort();


    charts.daily =
        new Chart(
            dailyChart,
            {

                type:"line",

                data:{

                    labels:
                        dates.map(
                            x =>
                                formatDate(x)
                        ),

                    datasets:[{

                        label:
                            "Full Day Present",

                        data:
                            dates.map(
                                x =>
                                    byDate[x]
                            ),

                        borderColor:
                            "#3157d5",

                        backgroundColor:
                            "rgba(49,87,213,.12)",

                        fill:true,

                        tension:.3

                    }]

                },

                options:{
                    responsive:true,
                    maintainAspectRatio:false
                }

/* =========================================================
   EMPLOYEE SUMMARY
========================================================= */

function employeeSummary(){

    const map =
        new Map();


    filteredData.forEach(
        function(x){

            const key =
                x.employeeId ||
                x.employeeName;


            if(!map.has(key)){

                map.set(
                    key,
                    {

                        id:
                            x.employeeId,

                        full:0,

                        morning:0,

                        afternoon:0,

                        absent:0,

                        morningLateMinutes:0,

                        afternoonLateMinutes:0,

                        lateMinutes:0

                    }
                );

            }


            const a =
                map.get(key);


            if(
                x.status ===
                "FULL DAY PRESENT"
            ){

                a.full++;

            }


            if(
                x.status ===
                "PRESENT IN THE MORNING"
            ){

                a.morning++;

            }


            if(
                x.status ===
                "PRESENT IN THE AFTERNOON"
            ){

                a.afternoon++;

            }


            if(
                x.status === "ABSENT"
            ){

                a.absent++;

            }


            a.morningLateMinutes +=
                x.morningLateMinutes || 0;


            a.afternoonLateMinutes +=
                x.afternoonLateMinutes || 0;


            a.lateMinutes +=
                x.lateMinutes || 0;

        }
    );


    return [
        ...map.values()
    ]
    .map(
        function(x){

            return {

                ...x,

                presentEquivalent:
                    x.full
                    +
                    (
                        x.morning +
                        x.afternoon
                    ) * 0.5

            };

        }
    );

}

/* =========================================================
   DATE REPORT
========================================================= */

function renderDates(){

    const map =
        new Map();


    filteredData.forEach(
        function(x){

            if(!x.date){
                return;
            }


            if(!map.has(x.date)){

                map.set(
                    x.date,
                    {

                        day:
                            x.day,

                        full:0,

                        morning:0,

                        afternoon:0,

                        absent:0,

                        off:0,

                        lateEmp:
                            new Set(),

                        morningLate:0,

                        afternoonLate:0,

                        lateMinutes:0

                    }
                );

            }


            const a =
                map.get(x.date);


            if(
                x.status ===
                "FULL DAY PRESENT"
            ){

                a.full++;

            }


            if(
                x.status ===
                "PRESENT IN THE MORNING"
            ){

                a.morning++;

            }


            if(
                x.status ===
                "PRESENT IN THE AFTERNOON"
            ){

                a.afternoon++;

            }


            if(
                x.status === "ABSENT"
            ){

                a.absent++;

            }


            if(
                x.status ===
                "SUNDAY / WEEKLY OFF"
            ){

                a.off++;

            }


            if(
                x.lateMinutes > 0
            ){

                a.lateEmp.add(
                    x.employeeId ||
                    x.employeeName
                );

            }


            a.morningLate +=
                x.morningLateMinutes || 0;


            a.afternoonLate +=
                x.afternoonLateMinutes || 0;


            a.lateMinutes +=
                x.lateMinutes || 0;

        }
    );


    const arr =
        [
            ...map.entries()
        ]
        .sort(
            function(a,b){

                return a[0].localeCompare(
                    b[0]
                );

            }
        );


    dateRows.innerHTML =
        arr.length
        ? arr.map(
            function(entry){

                const d =
                    entry[0];

                const x =
                    entry[1];


                return (

                    '<tr>'

                    + '<td>'
                    + formatDate(d)
                    + '</td>'

                    + '<td>'
                    + escapeHtml(x.day)
                    + '</td>'

                    + '<td>'
                    + x.full
                    + '</td>'

                    + '<td>'
                    + x.morning
                    + '</td>'

                    + '<td>'
                    + x.afternoon
                    + '</td>'

                    + '<td>'
                    + x.absent
                    + '</td>'

                    + '<td>'
                    + x.off
                    + '</td>'

                    + '<td>'
                    + x.lateEmp.size
                    + '</td>'

                    + '<td>'
                    + x.morningLate
                    + '</td>'

                    + '<td>'
                    + x.afternoonLate
                    + '</td>'

                    + '<td>'
                    + x.lateMinutes
                    + '</td>'

                    + '</tr>'

                );

            }
        ).join("")

        :

        '<tr>' +
        '<td colspan="12" class="empty">' +
        'No data to display.' +
        '</td>' +
        '</tr>';

}

function exportFilteredCSV(){

    if(!filteredData.length){

        alert(
            "No filtered data available to export."
        );

        return;

    }


    const headers = [

        "Date",

        "Day",

        "Employee ID",

        "Employee Name",

        "Attendance Status",

        "Attendance Value",

        "First Login",

        "Afternoon Check-out",

        "Afternoon Check-in",

        "Last Logout",

        "Punch Count",

        "Morning Late Minutes",

        "Afternoon Late Minutes",

        "Total Late Minutes"

    ];


    const rows = [

        headers,

        ...filteredData.map(
            function(x){

                return [

                    formatDate(
                        x.date
                    ),

                    x.day,

                    x.employeeId,

                    x.employeeName,

                    x.status,

                    x.attendanceValue,

                    x.firstLogin,

                    x.afternoonCheckout,

                    x.afternoonCheckin,

                    x.lastLogout,

                    x.punchCount,

                    x.morningLateMinutes,

                    x.afternoonLateMinutes,

                    x.lateMinutes

                ];

            }
        )

    ];


    const csv =
        rows.map(
            function(row){

                return row.map(
                    function(v){

                        return '"'
                            +
                            String(
                                v ?? ""
                            )
                            .replaceAll(
                                '"',
                                '""'
                            )
                            +
                            '"';

                    }
                ).join(",");

            }
        ).join("\n");


    const blob =
        new Blob(
            [csv],
            {
                type:"text/csv"
            }
        );


    const url =
        URL.createObjectURL(blob);


    const a =
        document.createElement("a");


    a.href =
        url;


    a.download =
        "attendance_report.csv";


    document.body.appendChild(a);

    a.click();

    document.body.removeChild(a);


    URL.revokeObjectURL(url);

}
